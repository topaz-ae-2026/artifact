```python
def is_valid_triangle(a: int, b: int, c: int) -> bool:
    return a + b > c and a + c > b and b + c > a

def triangle(a: int, b: int, c: int) -> str:
    if not is_valid_triangle(a, b, c):
        return "invalid"
    
    if a == b == c:
        return "equilateral"
    elif a == b or b == c or a == c:
        return "isosceles"
    else:
        return "scalene"
```
