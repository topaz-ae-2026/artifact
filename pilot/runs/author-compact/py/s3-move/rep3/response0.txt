```python
from dataclasses import dataclass


@dataclass
class Point:
    x: int
    y: int


def move(dir: str, x: int, y: int) -> Point:
    if dir == "N":
        return Point(x, y + 1)
    elif dir == "S":
        return Point(x, y - 1)
    elif dir == "E":
        return Point(x + 1, y)
    elif dir == "W":
        return Point(x - 1, y)
    else:
        return Point(x, y)
```
